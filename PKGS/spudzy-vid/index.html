Integrate Spudzy Vid properly

Load Three.js as an ES module, assign it to window.THREE, then load your hosted Spudzy Vid script. Always pass a real canvas and keep a spinner visible until RealLifeVideo.generate() resolves.

<script type="module">
  import * as THREE from "https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.module.js";
  window.THREE = THREE;
  await new Promise((resolve, reject) => {
    const s = document.createElement("script");
    s.src = "https://alaricholt677.github.io/PKGS/spudzy-vid/spudzy-vid-js.js";
    s.onload = resolve;
    s.onerror = reject;
    document.head.appendChild(s);
  });
</script>
<canvas id="genCanvas" width="960" height="540"></canvas>
<div id="preview"></div>
<script>
async function makeVideo(prompt) {
  const preview = document.getElementById("preview");
  const canvas = document.getElementById("genCanvas");
  preview.innerHTML = "<div class='spinner'></div> Generating...";
  const data = await RealLifeVideo.generate(prompt, {
    canvas,
    width: 960,
    height: 540,
    seconds: 6,
    fps: 30,
    renderer: "auto",
    gameplayType: "auto",
    audio: true,
    soundEffects: true,
    typedAudio: true
  });
  const blob = data.blob || (data.url && data.url.startsWith("blob:") ? await fetch(data.url).then(r => r.blob()) : null);
  if (!blob) throw new Error(data.reason || "No video blob returned");
  const url = URL.createObjectURL(blob);
  preview.innerHTML = "";
  const video = document.createElement("video");
  video.src = url;
  video.controls = true;
  video.playsInline = true;
  preview.appendChild(video);
}
</script>
